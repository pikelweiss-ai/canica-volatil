    const int SR = 22050;
    static Random R = new Random(1234);
    public class Lp { double y; public double Do(double x, double fc) { double a = 1 - Math.Exp(-2 * Math.PI * fc / SR); y += a * (x - y); return y; } }
    public class Osc { double ph; public double Sin(double f) { ph += 2 * Math.PI * f / SR; return Math.Sin(ph); } public double Saw(double f) { ph += f / SR; ph -= Math.Floor(ph); return 2 * ph - 1; } public double Sq(double f) { ph += f / SR; ph -= Math.Floor(ph); return ph < 0.5 ? 1 : -1; } }
    static double N() { return R.NextDouble() * 2 - 1; }
    static double Env(double t, double d, double a, double r) { double v = 1; if (t < a) v = t / a; double rem = d - t; if (rem < r) v *= Math.Max(0, rem / r); return v; }
    static double Dec(double t, double k) { return Math.Exp(-t * k); }
    static float[] Gen(double dur, Func<double, double> f) { int n = (int)(dur * SR); var b = new float[n]; for (int i = 0; i < n; i++) b[i] = (float)f(i / (double)SR); return b; }
    static void Add(float[] dst, float[] src, double at, double gain) { int o = (int)(at * SR); for (int i = 0; i < src.Length && o + i < dst.Length; i++) dst[o + i] += (float)(src[i] * gain); }
    static void Save(string path, float[] b)
    {
        float peak = 0; foreach (var v in b) peak = Math.Max(peak, Math.Abs(v)); float g = peak > 0 ? 0.85f / peak : 1;
        int fade = (int)(0.008 * SR);
        using (var w = new BinaryWriter(File.Create(path)))
        {
            int n = b.Length; w.Write(new[] { 'R', 'I', 'F', 'F' }); w.Write(36 + n * 2); w.Write(new[] { 'W', 'A', 'V', 'E', 'f', 'm', 't', ' ' }); w.Write(16); w.Write((short)1); w.Write((short)1); w.Write(SR); w.Write(SR * 2); w.Write((short)2); w.Write((short)16); w.Write(new[] { 'd', 'a', 't', 'a' }); w.Write(n * 2);
            for (int i = 0; i < n; i++) { double v = b[i] * g; if (i < fade) v *= i / (double)fade; if (n - i < fade) v *= (n - i) / (double)fade; w.Write((short)(Math.Max(-1, Math.Min(1, v)) * 32000)); }
        }
    }
